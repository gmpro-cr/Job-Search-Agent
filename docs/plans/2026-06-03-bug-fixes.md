### Task 2: Fix Gemini sleep blocking scraper thread (Bug 3)

**Files:**
- Modify: `agent/llm.py` — `_call_gemini()` function, line ~62-83

**Context:**
`time.sleep(4.5)` inside `_call_gemini()` blocks the calling thread for 4.5 seconds after EVERY call. During batch scoring of 100 jobs, this wastes 7.5 minutes. The fix: track the timestamp of the last successful Gemini call as a module-level variable, and sleep only the *remaining* time needed to respect the 4s gap (≈ no sleep if the caller itself is slow).

**Step 1: Add module-level last-call tracker to `agent/llm.py`**

After the imports (around line 12), add:
```python
import time as _time

_last_gemini_call: float = 0.0  # epoch seconds of last successful Gemini call
_GEMINI_MIN_INTERVAL: float = 4.1  # seconds between calls (15 RPM = 4s; 0.1s buffer)
```

**Step 2: Rewrite `_call_gemini` to use the tracker**

Replace the current `_call_gemini` body with:
```python
def _call_gemini(prompt: str, system: str) -> str:
    """Call Gemini Flash via Google AI API (free tier, 15 RPM)."""
    global _last_gemini_call
    api_key = os.getenv("GEMINI_API_KEY", "")
    if not api_key:
        raise ValueError("GEMINI_API_KEY not set")

    # Respect 15 RPM: sleep only the remaining interval, not a fixed 4.5s
    elapsed = _time.time() - _last_gemini_call
    wait = _GEMINI_MIN_INTERVAL - elapsed
    if wait > 0:
        _time.sleep(wait)

    full_prompt = f"{system}\n\n{prompt}" if system else prompt
    resp = requests.post(
        f"{GEMINI_URL}?key={api_key}",
        headers={"Content-Type": "application/json"},
        json={"contents": [{"parts": [{"text": full_prompt}]}]},
        timeout=GEMINI_TIMEOUT,
    )
    resp.raise_for_status()
    data = resp.json()
    try:
        result = data["candidates"][0]["content"]["parts"][0]["text"].strip()
        _last_gemini_call = _time.time()  # record AFTER successful response
        return result
    except (KeyError, IndexError) as e:
        raise ValueError(f"Unexpected Gemini response: {data}") from e
```

Remove the old `import time` inside the function.

```python
def _build_scoring_rubric(cv_data: dict, preferences: dict) -> dict:
    """
    Build dynamic rubric sections from user's CV data and preferences.
    Returns dict with keys: target_roles_rubric, domain_rubric.
    """
    # --- Role rubric from user's preferred job titles ---
    titles = [t.strip() for t in (preferences.get("job_titles") or []) if t.strip()]
    if titles:
        primary = titles[0]
        others = ", ".join(titles[1:4]) if len(titles) > 1 else ""
        role_rubric = (
            f"   35 — Exact match: {primary}\n"
            f"   25 — Close variant: {others or 'senior/lead variant of ' + primary}\n"
            f"   12 — Adjacent: related function with strong skill overlap\n"
            f"    4 — Wrong function but partial skill overlap\n"
            f"    0 — Unrelated function"
        )
    else:
        role_rubric = (
            "   35 — Exact title match to candidate's target role\n"
            "   25 — Close adjacent role with same core function\n"
            "   12 — Related role with meaningful skill overlap\n"
            "    4 — Wrong function but some transferable skills\n"
            "    0 — Completely unrelated function"
        )

    # --- Domain rubric from CV skills + preferences ---
    skills = [s.lower() for s in (cv_data.get("skills") or [])]
    raw_text = (cv_data.get("raw_text") or "").lower()

    # Detect domain from skills and raw CV text
    domain_signals = {
        "Fintech / Banking / Payments / Lending / NBFC / Credit / Insurance": [
            "fintech", "banking", "payments", "lending", "nbfc", "credit", "upi",
            "insurance", "wealth", "neobank", "fixed income", "treasury",
        ],
        "B2B SaaS / AI / ML / Data platform": [
            "saas", "machine learning", "llm", "genai", "data platform", "b2b",
            "api", "microservices", "mlops",
        ],
        "E-commerce / Edtech / Healthtech / Consumer tech": [
            "e-commerce", "ecommerce", "edtech", "healthtech", "consumer",
            "marketplace", "d2c",
        ],
        "Manufacturing / FMCG / Logistics / Ops": [
            "manufacturing", "fmcg", "logistics", "supply chain", "operations",
            "procurement",
        ],
    }

    top_domain = None
    top_count = 0
    for domain, keywords in domain_signals.items():
        count = sum(1 for kw in keywords if kw in skills or kw in raw_text)
        if count > top_count:
            top_count = count
            top_domain = domain

    if top_domain and top_count >= 2:
        others_domains = [d for d in domain_signals if d != top_domain]
        domain_rubric = (
            f"   25 — {top_domain}\n"
            f"   18 — {others_domains[0]}\n"
            f"   10 — {others_domains[1]}\n"
            f"    4 — {others_domains[2]}\n"
            f"    0 — Completely unrelated domain"
        )
    else:
        domain_rubric = (
            "   25 — Domain matches candidate's primary background closely\n"
            "   18 — Related domain with transferable knowledge\n"
            "   10 — Adjacent domain, some overlap\n"
            "    4 — Distant domain, minimal overlap\n"
            "    0 — Completely unrelated domain"
        )

    return {"target_roles_rubric": role_rubric, "domain_rubric": domain_rubric}
```

Replace the current `_startup_catchup()` with:
```python
def _startup_catchup():
    """
    On startup, fire any missed daily emails if we're past their scheduled time
    and the email hasn't been recorded as sent in the scheduler state.
    Reads the same state file that _start_simple_scheduler persists.
    """
    import threading, time as _time, json as _j
    from datetime import datetime as _dt, date as _date

    _SCHED_STATE_FILE = os.path.join(BASE_DIR, "data", "scheduler_state.json")

    def _load_state() -> dict:
        try:
            with open(_SCHED_STATE_FILE) as f:
                return _j.load(f)
        except Exception:
            return {}

    def _run():
        _time.sleep(5)  # wait for scheduler to settle
        now = _dt.now()
        today_str = _date.today().isoformat()
        state = _load_state()

        # PRD email at 08:00
        if now.hour >= 8 and state.get("prd") != today_str:
            logger.info("Startup catch-up: PRD email not sent today — sending now")
            try:
                _send_prd_email_job()
            except Exception as e:
                logger.error("Startup PRD catch-up failed: %s", e)
        else:
            logger.info("Startup catch-up: PRD already sent today or before 08:00 — skipping")

        # HR emails at 11:00
        if now.hour >= 11:
            from reminder_runner import load_reminders
            reminders = load_reminders()
            for r in reminders:
                if not r.get("hr_email_enabled"):
                    continue
                rid = r.get("id", "")
                last_hr_sent = r.get("last_hr_sent", "")
                if last_hr_sent and last_hr_sent[:10] == today_str:
                    continue
                logger.info("Startup catch-up: HR email for '%s' not sent today — sending now", r.get("name", rid))
                try:
                    _run_hr_email(rid)
                except Exception as e:
                    logger.error("Startup HR catch-up failed for %s: %s", rid, e)
        else:
            logger.info("Startup catch-up: before 11:00 — HR emails will fire on schedule")

    threading.Thread(target=_run, daemon=True).start()
```
